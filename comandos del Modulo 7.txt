# =========================
# Practica 1: compartir archivos entre linux utilizando NFS
# =========================

# Servidor principal
sudo apt update
# Actualizar repositorios

sudo apt install nfs-kernel-server -y
# Instalar servidor NFS

sudo mkdir -p /srv/OS3
# Crear carpeta compartida

cd /srv/OS3
# Entrar a la carpeta compartida

sudo bash -c 'for i in {1..100}; do touch /srv/OS3/Adrian$i.txt; done'
# Crear 100 archivos de prueba

ls | wc -l
# Contar cantidad de archivos

sudo chown nobody:nogroup /srv/OS3
# Cambiar propietario recomendado para NFS

sudo chmod 755 /srv/OS3
# Asignar permisos a la carpeta

sudo nano /etc/exports
# Editar archivo de exportaciones NFS

/srv/nfs/OS3 tu_ip_vmCliente(rw,sync,no_subtree_check)

sudo exportfs -a
# Aplicar configuración NFS

sudo exportfs -v
# Verificar exportaciones activas

sudo systemctl restart nfs-server
# Reiniciar servicio NFS

sudo systemctl enable nfs-server
# Habilitar NFS al iniciar

sudo ufw allow from 192.168.1.0/24 to any port nfs
# Permitir NFS en la red local

sudo ufw allow 2049
# Permitir puerto NFS

sudo ufw reload
# Recargar firewall

# Servidor cliente
sudo apt update
# Actualizar repositorios

sudo apt install nfs-common -y
# Instalar cliente NFS

sudo mkdir -p /mnt/OS3
# Crear punto de montaje

sudo mount 192.168.1.50:/srv/OS3 /mnt/OS3
# Montar recurso compartido

df -h
# Verificar montaje

ls /mnt/OS3
# Ver contenido montado

# MONTAJE AUTOMÁTICO
sudo nano /etc/fstab
# Configurar montaje automático

192.168.1.50:/srv/OS3 /mnt/OS3 nfs defaults 0 0

sudo umount /mnt/OS3
# Desmontar recurso

sudo mount -a
# Montar todo lo configurado en fstab

ls /mnt/OS3
# Verificar contenido

sudo reboot
# Reiniciar máquina cliente

ls /mnt/OS3
# Confirmar persistencia del montaje


# =========================
# Practica 2: Creacion de fileserver compatible con Windows utilizando SAMBA
# =========================

sudo apt update && sudo apt install samba -y
# Instalar Samba

sudo mkdir -p /srv/samba/OS3
# Crear carpeta compartida

sudo groupadd smbgrupo
# Crear grupo para Samba

sudo useradd -m -G smbgrupo sistemasO3
# Crear usuario del recurso compartido

sudo passwd sistemasO3
# Asignar contraseña Linux

sudo smbpasswd -a sistemasO3
# Crear usuario Samba

sudo smbpasswd -e sistemasO3
# Habilitar usuario Samba

sudo chown -R root:smbgrupo /srv/samba/OS3
# Cambiar grupo propietario

sudo chmod -R 770 /srv/samba/OS3
# Asignar permisos completos al grupo

sudo usermod -aG smbgrupo sistemas3
# Agregar usuario local al grupo

newgrp smbgrupo
# Aplicar grupo actual sin reiniciar sesión

sudo bash -c 'for i in {1..100}; do touch /srv/samba/OS3/adrian$i.txt; done'
# Crear 100 archivos de prueba

sudo nano /etc/samba/smb.conf
# Editar configuración Samba

# Dentro del archivo agrega esto al final:
[OS3]
   path = /srv/samba/OS3
   browseable = yes
   read only = no
   writable = yes
   valid users = @smbgrupo
   create mask = 0770
   directory mask = 0770
   force group = smbgrupo

# Y en [global] tener:
server min protocol = SMB2
client min protocol = SMB2

# Luego continúa:
testparm
# Validar configuración

sudo systemctl restart smbd
# Reiniciar Samba

sudo systemctl status smbd
# Ver estado del servicio

smbclient -L localhost -U sistemasO3
# Listar recursos compartidos

ip a
# Ver IP del servidor

sudo ufw disable
# Desactivar firewall para pruebas

# Usuario:
# sistemasO3


# =========================
# Practica 3: Creacion de controlador de Dominio con cliente Windows
# =========================

sudo hostnamectl set-hostname dc
# Cambiar hostname del controlador de dominio

sudo nano /etc/hosts
# Configurar resolución local

# CONTENIDO:
127.0.0.1    localhost
192.168.1.5 dc.so3.inet dc

hostname -f
# Verificar FQDN

ping -c2 dc.so3.inet
# Probar resolución

sudo systemctl disable --now systemd-resolved
# Desactivar resolved

sudo unlink /etc/resolv.conf
# Eliminar enlace simbólico

sudo nano /etc/resolv.conf
# Crear DNS manual

# CONTENIDO:
nameserver 192.168.100.14
nameserver 8.8.8.8
search so3.inet

sudo chattr +i /etc/resolv.conf
# Proteger resolv.conf

sudo apt update
# Actualizar repositorios

sudo apt install -y acl attr samba samba-dsdb-modules samba-vfs-modules smbclient winbind libpam-winbind libnss-winbind libpam-krb5 krb5-config krb5-user dnsutils chrony net-tools
# Instalar dependencias para AD DC

# EN PANTALLA KERBEROS
# Poner so3.inet

# Deshabilitar servicios de Samba que no se usan en AD DC
sudo systemctl disable --now smbd nmbd winbind

sudo systemctl unmask samba-ad-dc
# Habilitar servicio AD DC

sudo systemctl enable --now samba-ad-dc
# Iniciar servicio AD DC

sudo mv /etc/samba/smb.conf /etc/samba/smb.conf.orig
# Respaldar smb.conf

sudo samba-tool domain provision
# Aprovisionar dominio

realm=SO3.INET
domain=SO3
server-role=dc
dns-backend=SAMBA_INTERNAL
dns-forwarder=8.8.8.8
adminpass='Admin123'

sudo mv /etc/krb5.conf /etc/krb5.conf.orig
# Respaldar Kerberos

sudo cp /var/lib/samba/private/krb5.conf /etc/krb5.conf
# Copiar configuración Kerberos

# Reiniciar Samba AD DC
sudo systemctl restart samba-ad-dc

sudo systemctl status samba-ad-dc
# Verificar servicio

sudo chown root:_chrony /var/lib/samba/ntp_signd/
# Asignar permisos Chrony

sudo chmod 750 /var/lib/samba/ntp_signd/
# Ajustar permisos

sudo nano /etc/chrony/chrony.conf
# Configurar Chrony para AD

# AGREGAR AL FINAL:
# Samba AD-DC NTP
bindcmdaddress 192.168.1.5
allow        192.168.1.0/24
ntpsigndsocket /var/lib/samba/ntp_signd

sudo systemctl restart chronyd
# Reiniciar Chrony

sudo systemctl status chronyd
# Ver estado

# Comprobar resolución DNS interna
host -t A dc.so3.inet

# lista de recursos compartidos de Samba
smbclient -L so3.inet -N

# Autenticarse en Kerberos con el usuario administrador del dominio
kinit administrator@SO3.INET

# ticket Kerberos
klist

# carpeta netlogon
sudo smbclient //localhost/netlogon -U 'administrator'

sudo samba-tool user create lanegracubana Lord20250744 --login-shell=/bin/bash
# Crear usuario del dominio

sudo samba-tool user list
